
local code = ""

function missing(t, f, fallback)
    if type(f) == t then return f end
    return fallback
end
queueteleport = missing("function", queue_on_teleport or (syn and syn.queue_on_teleport) or (fluxus and fluxus.queue_on_teleport))

local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local TeleportService = game:GetService("TeleportService")
local HttpService = game:GetService("HttpService")

local PLACE_ID = game.PlaceId
local plr = Players.LocalPlayer
local teleporting = false


local function serverHop()
    if queueteleport and code ~= "" then
        queueteleport(code)
    end

    local cursor = ""
    local servers = {}

    repeat
        local url = "https://games.roblox.com/v1/games/" .. PLACE_ID ..
            "/servers/Public?sortOrder=Asc&limit=100&cursor=" .. cursor

        local data = HttpService:JSONDecode(game:HttpGet(url))

        for _, server in ipairs(data.data) do
            if server.playing < server.maxPlayers then
                table.insert(servers, server)
            end
        end

        cursor = data.nextPageCursor or ""
    until #servers > 0 or cursor == ""

    if #servers > 0 then
        local server = servers[math.random(1, #servers)]
        TeleportService:TeleportToPlaceInstance(
            PLACE_ID,
            server.id,
            plr
        )
    end
end

function TP1(targetCFrame)
    local char = plr.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp or not targetCFrame then return end

    teleporting = true  

    local part = Instance.new("Part")  
    part.Name = "PartTele"  
    part.Size = Vector3.new(10, 1, 10)  
    part.Anchored = true  
    part.Transparency = 1  
    part.CanCollide = true  
    part.CFrame = hrp.CFrame  
    part.Parent = char  

    local distance = (targetCFrame.Position - hrp.Position).Magnitude  
    local tween = TweenService:Create(  
        part,  
        TweenInfo.new(distance / 360, Enum.EasingStyle.Linear),  
        {CFrame = targetCFrame}  
    )  

    local connection  
    connection = part:GetPropertyChangedSignal("CFrame"):Connect(function()  
        if teleporting and hrp.Parent then  
            hrp.CFrame = part.CFrame  
        end  
    end)  

    tween:Play()  
    tween.Completed:Wait()  

    teleporting = false  
    if connection then connection:Disconnect() end  
    if part then part:Destroy() end
end
